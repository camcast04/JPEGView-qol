// This file defines custom key mapping for JPEGView

// Map M to crop the selected area
M                       IDM_CROP_SEL

// Map Ctrl+Up to save without prompt
Ctrl+Up                 IDM_SAVE_ALLOW_NO_PROMPT

// Map Ctrl+Shift+Up to open save dialog (same as Ctrl+S)
Ctrl+Shift+Up           IDM_SAVE

// Map X to sort by modification date (original M function)
X                       IDM_SORT_MOD_DATE

// Restore original rotation shortcuts
Up                      IDM_ROTATE_270
Down                    IDM_ROTATE_90

2                       IDM_CROPMODE_4_3
3                       IDM_CROPMODE_3_2
4                       IDM_CROPMODE_16_9
5                       IDM_CROPMODE_IMAGE
9                       IDM_CROPMODE_FIXED_SIZE
